function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); var action = data.action; var urunId = String(data.id).trim(); var resimSonuc = data.resimler; // Resim varsa ve Base64 formatındaysa doğrudan Google Drive'a kaydet if (data.resimler && data.resimler.indexOf('base64') !== -1) { try { var resimDizisi = data.resimler.split('|'); var linkler = []; for (var r = 0; r < resimDizisi.length; r++) { if (resimDizisi[r].indexOf('base64,') !== -1) { var base64Data = resimDizisi[r].split('base64,')[1]; var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), 'image/jpeg', 'OzenOto_' + urunId + '_' + r + '.jpg'); var file = DriveApp.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); linkler.push("https://lh3.googleusercontent.com/d/" + file.getId()); } else { linkler.push(resimDizisi[r]); } } resimSonuc = linkler.join('|'); } catch(err) { resimSonuc = "dukkan.png"; } } if (action === "update") { var rows = sheet.getDataRange().getValues(); var bulundu = false; for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]).trim() === urunId) { sheet.getRange(i + 1, 2).setValue(data.baslik); sheet.getRange(i + 1, 3).setValue(data.arac); sheet.getRange(i + 1, 4).setValue(data.oem); if (resimSonuc && resimSonuc !== "") { sheet.getRange(i + 1, 5).setValue(resimSonuc); } sheet.getRange(i + 1, 6).setValue(data.etiket); sheet.getRange(i + 1, 7).setValue(data.aciklama); bulundu = true; break; } } if (!bulundu) { sheet.appendRow([urunId, data.baslik, data.arac, data.oem, resimSonuc || "dukkan.png", data.etiket, data.aciklama]); } } else { sheet.appendRow([urunId, data.baslik, data.arac, data.oem, resimSonuc || "dukkan.png", data.etiket, data.aciklama]); } return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON); } catch(hata) { return ContentService.createTextOutput(JSON.stringify({"error": hata.message})).setMimeType(ContentService.MimeType.JSON); } }
